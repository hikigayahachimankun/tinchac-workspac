## Lệnh tái kiểm offline

Chạy tại root repo; dùng model T01 hiện có, không gọi mạng hoặc đọc gold:

```bash
PYTHONPATH=. .venv/bin/python - <<'PY'
import hashlib,json,unicodedata
from pathlib import Path
from collections import Counter
from backend.app.models import SourceVersion
real=json.loads(Path('data/corpus/sources.json').read_text())
fixture=[json.loads(s) for s in Path('data/fixtures/sources.jsonl').read_text().splitlines() if s]
for group,is_fixture in [(real,False),(fixture,True)]:
 for s in group:
  SourceVersion.model_validate(s)
  assert s['is_fixture'] is is_fixture
  assert s['text']==unicodedata.normalize('NFC',s['text']) and '\r' not in s['text']
  digest=hashlib.sha256(s['text'].encode()).hexdigest()
  assert s['content_sha256']==digest
  assert s['version_id'].endswith(digest[:16])
  if not is_fixture:
   assert s['source_id']=='dut_'+hashlib.sha256(s['canonical_url'].encode()).hexdigest()[:16]
   assert s['available_at']==s['retrieved_at'] and s['available_basis']=='first_seen'
  if s['review_status']=='quarantined':assert not s['facts']
  for f in s['facts']:assert s['text'][f['span']['start']:f['span']['end']]==f['quote']
assert len({s['version_id'] for s in real+fixture})==len(real+fixture)
assert not ({s['family_id'] for s in real}&{s['family_id'] for s in fixture})
assert len(real)==6 and len(fixture)==1
config=json.loads(Path('config/sources.yaml').read_text())
assert {s['version_id'] for s in config['sources']}=={s['version_id'] for s in real}
assert not Path('data/raw/t02/12381.pdf').exists()
assert not any(Path('data/raw/t02').glob('12381-p[23].png'))
print('SourceVersion v1: 6 real + 1 fixture valid; NFC, SHA-256, quote/spans, IDs, availability and separation pass')
print('Real:',dict(Counter(s['review_status'] for s in real)),'facts:',sum(len(s['facts']) for s in real))
PY
.venv/bin/python -m scripts.export_contracts --check
git diff --check
```
